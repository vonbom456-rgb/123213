cmake_minimum_required(VERSION 3.15)
project(TribeLeaderboard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED ARKAPI_ROOT)
  message(FATAL_ERROR "ARKAPI_ROOT is required")
endif()
add_library(TribeLeaderboard SHARED Source/TribeLeaderboard.cpp)
target_compile_definitions(TribeLeaderboard PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(TribeLeaderboard PRIVATE "${ARKAPI_ROOT}/version/Core/Public" "${ARKAPI_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/Source")
target_link_directories(TribeLeaderboard PRIVATE "${ARKAPI_ROOT}/out_lib")
target_link_libraries(TribeLeaderboard PRIVATE ArkApi)
