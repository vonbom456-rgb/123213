cmake_minimum_required(VERSION 3.15)
project(GathererRouter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED ARKAPI_ROOT)
  message(FATAL_ERROR "ARKAPI_ROOT is required")
endif()
add_library(GathererRouter SHARED Source/GathererRouter.cpp)
target_compile_definitions(GathererRouter PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(GathererRouter PRIVATE "${ARKAPI_ROOT}/version/Core/Public" "${ARKAPI_ROOT}/include")
target_link_directories(GathererRouter PRIVATE "${ARKAPI_ROOT}/out_lib")
target_link_libraries(GathererRouter PRIVATE ArkApi)
